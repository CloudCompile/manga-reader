    63→// ─── LocalStorage helpers ────────────────────────────
    64→function getHistory(): HistoryEntry[] {
    65→  try { return JSON.parse(localStorage.getItem("manga-history") || "[]"); } catch { return []; }
    66→}
    67→function saveHistory(entry: HistoryEntry) {
    68→  const h = getHistory().filter((e) => e.url !== entry.url);
    69→  h.unshift({ ...entry, timestamp: Date.now() });
    70→  localStorage.setItem("manga-history", JSON.stringify(h.slice(0, 50)));
    71→}
    72→function getLastRead(): string | null {
    73→  return localStorage.getItem("manga-last-read");
    74→}
    75→function setLastRead(url: string) {
    76→  localStorage.setItem("manga-last-read", url);
    77→}

    79→// ─── IndexedDB for offline ───────────────────────────
    80→const DB_NAME = "manga-offline";
    81→const DB_VERSION = 1;
    82→const STORE = "chapters";
    83→
    84→function openDB(): Promise<IDBDatabase> {
    85→  return new Promise((resolve, reject) => {
    86→    const req = indexedDB.open(DB_NAME, DB_VERSION);
    87→    req.onupgradeneeded = () => req.result.createObjectStore(STORE, { keyPath: "url" });
    88→    req.onsuccess = () => resolve(req.result);
    89→    req.onerror = () => reject(req.error);
    90→  });
    91→}
    92→
    93→async function saveOffline(data: OfflineChapter) {
    94→  const db = await openDB();
    95→  return new Promise<void>((resolve, reject) => {
    96→    const tx = db.transaction(STORE, "readwrite");
    97→    tx.objectStore(STORE).put(data);
    98→    tx.oncomplete = () => resolve();
    99→    tx.onerror = () => reject(tx.error);
   100→  });
   101→}
   102→
   103→async function getOffline(url: string): Promise<OfflineChapter | null> {
   104→  const db = await openDB();
   105→  return new Promise((resolve, reject) => {
   106→    const tx = db.transaction(STORE, "readonly");
   107→    const req = tx.objectStore(STORE).get(url);
   108→    req.onsuccess = () => resolve(req.result || null);
   109→    req.onerror = () => reject(req.error);
   110→  });
   111→}
   112→
   113→async function getAllOffline(): Promise<OfflineChapter[]> {
   114→  const db = await openDB();
   115→  return new Promise((resolve, reject) => {
   116→    const tx = db.transaction(STORE, "readonly");
   117→    const req = tx.objectStore(STORE).getAll();
   118→    req.onsuccess = () => resolve(req.result || []);
   119→    req.onerror = () => reject(req.error);
   120→  });
   121→}
   122→
   123→async function deleteOffline(url: string) {
   124→  const db = await openDB();
   125→  return new Promise<void>((resolve, reject) => {
   126→    const tx = db.transaction(STORE, "readwrite");
   127→    tx.objectStore(STORE).delete(url);
   128→    tx.oncomplete = () => resolve();
   129→    tx.onerror = () => reject(tx.error);
   130→  });
   131→}

   152→  const [history, setHistory] = useState<HistoryEntry[]>([]);
   153→  const [showHistory, setShowHistory] = useState(false);

   155→  const [showOffline, setShowOffline] = useState(false);
   156→  const [currentPage, setCurrentPage] = useState(0);
   157→  const [pageInput, setPageInput] = useState("");

   161→  const [showMobileMenu, setShowMobileMenu] = useState(false);

   163→  const scrollRef = useRef<HTMLDivElement>(null);
   164→  const imgRefs = useRef<Map<number, HTMLImageElement>>(new Map());

   166→  // Init
   167→  useEffect(() => {
   168→    setHistory(getHistory());
   169→    getAllOffline().then(setOfflineChapters).catch(() => {});
   170→    const saved = getLastRead();
   171→    if (saved) setUrl(saved);
   172→  }, []);
   173→

   177→    if (!el) return;

   180→      // Track which page we're viewing

   183→        const rect = imgs[i].getBoundingClientRect();
   184→        if (rect.top >= -rect.height / 2) {
   185→          setCurrentPage(i + 1);
   186→          break;
   187→        }
   188→      }
   189→    };
   190→    el.addEventListener("scroll", onScroll, { passive: true });
   191→    return () => el.removeEventListener("scroll", onScroll);
   192→  }, [data]);
   193→
   194→  // Set initial page to 1 when data loads
   195→  useEffect(() => { if (data?.images.length) setCurrentPage(1); }, [data?.images.length]);
   196→
   197→  // Keyboard shortcuts

   199→    const handler = (e: KeyboardEvent) => {
   200→      if (e.target instanceof HTMLInputElement || e.target instanceof HTMLTextAreaElement) return;
   201→      if (e.key === "ArrowLeft" && data?.prevUrl) {
   202→        e.preventDefault();
   203→        goToChapter(data.prevUrl);
   204→      } else if (e.key === "ArrowRight" && data?.nextUrl) {
   205→        e.preventDefault();
   206→        goToChapter(data.nextUrl);
   207→      } else if (e.key === "t" || e.key === "T") {
   208→        toggleTheme();

   210→        setShowChapters((v) => !v);
   211→      } else if (e.key === "Escape") {

   213→        setShowSearch(false);
   214→        setShowHistory(false);
   215→        setShowOffline(false);
   216→        setShowPageJumper(false);
   217→        setShowMobileMenu(false);

   219→        setShowPageJumper((v) => !v);
   220→      }
   221→    };
   222→    window.addEventListener("keydown", handler);
   223→    return () => window.removeEventListener("keydown", handler);
   224→  }, [data]);
   225→

   227→    setLoadedImages((prev) => new Set(prev).add(index));
   228→  }, []);
   229→
   230→  const loadChapter = useCallback(async (chapterUrl: string, offline?: OfflineChapter) => {
   231→    if (!chapterUrl.trim()) return;
   232→    setLoading(true);
   233→    setError("");
   234→    setData(null);
   235→    setLoadedImages(new Set());

   237→    setShowScrollTop(false);
   238→    if (scrollRef.current) scrollRef.current.scrollTop = 0;
   239→    setUrl(chapterUrl);
   240→    setLastRead(chapterUrl);
   241→    setShowChapters(false);
   242→    setShowMobileMenu(false);
   243→
   244→    try {
   245→      if (offline) {
   246→        // Load from offline cache
   247→        const chapterData: ChapterData = {
   248→          title: `${offline.mangaName}, Chapter ${offline.chapterNum}`,
   249→          images: offline.images,
   250→          prevUrl: null,
   251→          nextUrl: null,
   252→          mangaName: offline.mangaName,
   253→          chapterNum: offline.chapterNum,
   254→          seriesSlug: "",
   255→          warning: "Offline mode — navigation disabled",
   256→        };

   258→        setIsOfflineMode(true);

   260→        setIsOfflineMode(false);
   261→        const res = await fetch("/api/scrape", {
   262→          method: "POST",
   263→          headers: { "Content-Type": "application/json" },
   264→          body: JSON.stringify({ url: chapterUrl }),
   265→        });
   266→        const json = await res.json();
   267→        if (!res.ok) throw new Error(json.error || "Failed to load chapter");
   268→        setData(json);
   269→        saveHistory({ url: chapterUrl, mangaName: json.mangaName, chapterNum: json.chapterNum, timestamp: Date.now() });
   270→        setHistory(getHistory());
   271→
   272→        // Fetch chapter list in background
   273→        if (json.seriesSlug) fetchChapterList(chapterUrl);
   274→      }
   275→    } catch (err) {
   276→      setError(err instanceof Error ? err.message : "Something went wrong");
   277→    } finally {
   278→      setLoading(false);
   279→    }
   280→  }, []);

   285→      const res = await fetch("/api/chapters", {
   286→        method: "POST",
   287→        headers: { "Content-Type": "application/json" },
   288→        body: JSON.stringify({ url: chapterUrl }),
   289→      });

   291→      if (res.ok) setChapters(json.chapters || []);
   292→    } catch {
   293→      // Silent fail for chapter list
   294→    } finally {

   296→    }
   297→  };

   303→      const res = await fetch(`/api/search?q=${encodeURIComponent(searchQuery)}`);
   304→      const json = await res.json();
   305→      if (res.ok) setSearchResults(json);
   306→    } catch { /* ignore */ }
   307→    finally { setSearchLoading(false); }
   308→  };
   309→

   314→      // Pre-fetch all images through proxy to cache them

   318→        mangaName: data.mangaName,
   319→        chapterNum: data.chapterNum,
   320→        images: imageUrls,
   321→        savedAt: Date.now(),
   322→      });

   325→    } catch { /* ignore */ }

   331→  const scrollToTop = () => scrollRef.current?.scrollTo({ top: 0, behavior: "smooth" });
   332→  const toggleTheme = () => {
   333→    const next = theme === "dark" ? "light" : "dark";
   334→    setTheme(next);
   335→    localStorage.setItem("manga-theme", next);
   336→  };

   341→    setShowPageJumper(false);

   345→  const totalPages = data?.images.length || 0;
   346→  const displayImages = data ? data.images.map(PROXY) : [];
   347→  const isDark = theme === "dark";
   348→
   349→  const bg = isDark ? "bg-[#0a0a0a]" : "bg-[#f5f5f5]";
   350→  const textCol = isDark ? "text-white" : "text-gray-900";
   351→  const subText = isDark ? "text-white/50" : "text-gray-500";
   352→  const barBg = isDark ? "bg-[#111] border-white/10" : "bg-white border-gray-200";
   353→  const panelBg = isDark ? "bg-[#111]" : "bg-white";
   354→  const hoverBg = isDark ? "hover:bg-white/10" : "hover:bg-gray-100";
   355→
   356→  const imgStyle: React.CSSProperties =
   357→    fitMode === "width" ? { width: "100%", height: "auto" }

   366→    return (
   367→      <div className="fixed inset-0 z-50 flex justify-end" onClick={onClose}>

   369→        <div className={`relative w-full max-w-sm ${panelBg} ${textCol} shadow-2xl flex flex-col`} style={{animation: "slideIn 200ms ease-out"}} onClick={(e) => e.stopPropagation()}>
   370→          <div className={`flex items-center justify-between px-4 py-3 border-b ${isDark ? "border-white/10" : "border-gray-200"}`}>
   371→            <div className="flex items-center gap-2">
   372→              {icon}
   373→              <h2 className="font-semibold text-sm">{title}</h2>
   374→            </div>
   375→            <button onClick={onClose} className={`p-1 rounded ${hoverBg}`}><X className="h-4 w-4" /></button>
   376→          </div>
   377→          <ScrollArea className="flex-1">{children}</ScrollArea>
   378→        </div>
   379→      </div>

   381→  };

   484→            {offlineChapters.map((o) => (
   485→              <div key={o.url} className="flex items-center justify-between px-3 py-2 rounded-lg hover:bg-white/5">
   486→                <button onClick={() => { loadChapter(o.url, o); setShowOffline(false); }} className="text-left text-sm text-white/80">
   487→                  <div className="font-medium">{o.mangaName}</div>
   488→                  <div className="text-xs text-white/40">Ch {o.chapterNum} · {o.images.length} pages</div>
   489→                </button>
   490→                <button onClick={() => { deleteOffline(o.url); getAllOffline().then(setOfflineChapters); }}
   491→                  className="text-white/30 hover:text-red-400 p-1"><X className="h-3 w-3" /></button>
   492→              </div>
   493→            ))}
   494→          </div>
   495→        </SidePanel>
   496→      </div>
   497→    );
   498→  }
   499→
   500→  // ─── Error ──────────────────────────────────────────
   501→  if (error) {
   502→    return (
   503→      <div className="min-h-screen flex flex-col items-center justify-center gap-6 p-6 bg-[#0a0a0a]">
   504→        <p className="text-red-400 text-sm max-w-md text-center">{error}</p>
   505→        <form onSubmit={handleSubmit} className="w-full max-w-lg flex gap-2">
   506→          <Input type="url" placeholder="Try a different URL…" value={url} onChange={(e) => setUrl(e.target.value)}
   507→            className="bg-white/5 border-white/10 text-white placeholder:text-white/30" />
   508→          <Button type="submit">Retry</Button>
   509→        </form>
   510→        {data?.prevUrl && <Button variant="outline" onClick={() => goToChapter(data.prevUrl)}><ChevronLeft className="h-4 w-4 mr-1" /> Previous Chapter</Button>}
   511→      </div>
   512→    );
   513→  }
   514→
   515→  // ─── Reader ─────────────────────────────────────────
   516→  return (
   517→    <div className={`h-screen flex flex-col ${bg} ${textCol} overflow-hidden`}>
   518→      {/* Top bar */}
   519→      <header className={`shrink-0 border-b px-3 py-2 flex items-center gap-2 ${barBg} z-40`}>
   520→        <button onClick={() => setShowMobileMenu((v) => !v)} className={`p-1.5 rounded sm:hidden ${hoverBg}`}><Menu className="h-5 w-5" /></button>
   521→        <div className="flex items-center gap-2 min-w-0 flex-1">
   522→          <BookOpen className="h-4 w-4 shrink-0" />
   523→          <h1 className="text-sm font-semibold truncate">{data.mangaName}</h1>
   524→          {data.chapterNum && <span className={subText}>· Ch {data.chapterNum}</span>}
   525→          {isOfflineMode && <span className="text-[10px] px-1.5 py-0.5 rounded bg-yellow-500/20 text-yellow-400">OFFLINE</span>}
   526→        </div>
   527→
   528→        {/* Desktop buttons */}
   529→        <div className="hidden sm:flex items-center gap-1">
   530→          <button onClick={() => setShowChapters((v) => !v)} className={`px-2 py-1 rounded text-xs transition-colors ${showChapters ? "bg-white/15" : ""} ${subText} ${hoverBg}`} title="Chapter list (L)"><List className="h-3.5 w-3.5 inline mr-1" />Chapters</button>
   531→          <div className="w-px h-4 bg-white/10 mx-1" />
   532→          {(["width", "height", "original"] as const).map((fm) => (
   533→            <button key={fm} onClick={() => setFitMode(fm)}
   534→              className={`px-2 py-1 rounded text-xs transition-colors ${fitMode === fm ? "bg-white/15" : ""} ${subText} ${hoverBg}`}>
   535→              {fm === "width" ? "Fit W" : fm === "height" ? "Fit H" : "1:1"}
   536→            </button>
   537→          ))}
   538→          <div className="w-px h-4 bg-white/10 mx-1" />
   539→          <button onClick={() => setShowPageJumper((v) => !v)} className={`px-2 py-1 rounded text-xs ${subText} ${hoverBg}`} title="Jump to page (G)">
   540→            {currentPage}/{totalPages}
   541→          </button>
   542→          <button onClick={handleSaveOffline} disabled={savingOffline} className={`px-2 py-1 rounded text-xs ${subText} ${hoverBg}`} title="Save offline">
   543→            {savingOffline ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Download className="h-3.5 w-3.5" />}
   544→          </button>
   545→          <button onClick={toggleTheme} className={`p-1.5 rounded ${hoverBg} ${subText}`}><Sun className="h-4 w-4" /></button>
   546→        </div>
   547→      </header>
   548→
   549→      {/* Mobile dropdown menu */}
   550→      {showMobileMenu && (
   551→        <div className={`sm:hidden border-b px-3 py-3 flex flex-wrap gap-2 ${barBg} z-30`}>
   552→          <button onClick={() => { setShowChapters((v) => !v); setShowMobileMenu(false); }} className={`px-3 py-1.5 rounded text-xs ${hoverBg}`}><List className="h-3.5 w-3.5 inline mr-1" />Chapters</button>
   553→          {(["width", "height", "original"] as const).map((fm) => (
   554→            <button key={fm} onClick={() => { setFitMode(fm); setShowMobileMenu(false); }}
   555→              className={`px-3 py-1.5 rounded text-xs ${fitMode === fm ? "bg-white/15" : ""} ${hoverBg}`}>
   556→              {fm === "width" ? "Fit Width" : fm === "height" ? "Fit Height" : "Original"}
   557→            </button>
   558→          ))}
   559→          <button onClick={() => { setShowPageJumper((v) => !v); setShowMobileMenu(false); }} className={`px-3 py-1.5 rounded text-xs ${hoverBg}`}>Page {currentPage}/{totalPages}</button>
   560→          <button onClick={() => { handleSaveOffline(); setShowMobileMenu(false); }} disabled={savingOffline} className={`px-3 py-1.5 rounded text-xs ${hoverBg}`}>
   561→            {savingOffline ? "Saving…" : <><Download className="h-3.5 w-3.5 inline mr-1" />Save</>}
   562→          </button>
   563→          <button onClick={() => { toggleTheme(); setShowMobileMenu(false); }} className={`px-3 py-1.5 rounded text-xs ${hoverBg}`}>
   564→            {isDark ? <Sun className="h-3.5 w-3.5 inline mr-1" /> : <Moon className="h-3.5 w-3.5 inline mr-1" />}Theme
   565→          </button>
   566→        </div>
   567→      )}
   568→
   569→      {/* Page jumper overlay */}
   570→      {showPageJumper && (
   571→        <div className="absolute top-16 left-1/2 -translate-x-1/2 z-50 flex items-center gap-2 p-3 rounded-lg shadow-xl border"
   572→          style={{ backgroundColor: isDark ? "#1a1a1a" : "#fff", borderColor: isDark ? "rgba(255,255,255,0.1)" : "#e5e5e5" }}>
   573→          <span className={`text-xs ${subText}`}>Jump to:</span>
   574→          <input type="number" min={1} max={totalPages} value={pageInput}
   575→            onChange={(e) => setPageInput(e.target.value)}
   576→            onKeyDown={(e) => { if (e.key === "Enter") jumpToPage(parseInt(pageInput)); }}
   577→            placeholder={`1-${totalPages}`}
   578→            className={`w-20 px-2 py-1 rounded text-sm ${isDark ? "bg-white/10 text-white" : "bg-gray-100 text-gray-900"}`}
   579→            autoFocus />
   580→          <Button size="sm" onClick={() => jumpToPage(parseInt(pageInput))} disabled={!pageInput}>Go</Button>
   581→        </div>
   582→      )}
   583→
   584→      {/* Chapter sidebar */}
   585→      <SidePanel title={`Chapters (${chapters.length})`} icon={<List className="h-4 w-4" />} show={showChapters} onClose={() => setShowChapters(false)}>
   586→        <div className="p-3">
   587→          {chaptersLoading && <div className="flex justify-center py-8"><Loader2 className="h-5 w-5 animate-spin" /></div>}
   588→          {chapters.length === 0 && !chaptersLoading && <p className="text-white/40 text-sm text-center py-8">Load a chapter first to see the list</p>}
   589→          <div className="space-y-0.5 max-h-[70vh] overflow-y-auto">
   590→            {chapters.map((ch) => {
   591→              const isCurrent = ch.num === parseInt(data.chapterNum);
   592→              return (
   593→                <button key={ch.num} onClick={() => { loadChapter(ch.url); setShowChapters(false); }}
   594→                  className={`w-full text-left px-3 py-1.5 rounded text-sm transition-colors ${isCurrent
   595→                    ? "bg-white/15 font-semibold"
   596→                    : hoverBg
   597→                  } ${isCurrent ? textCol : subText}`}>
   598→                  Chapter {ch.num}
   599→                </button>
   600→              );
   601→            })}
   602→          </div>
   603→        </div>
   604→      </SidePanel>
   605→
   606→      {/* Image area */}
   607→      <div ref={scrollRef} className="flex-1 overflow-y-auto relative">
   608→        {data.warning && (
   609→          <div className="max-w-2xl mx-auto mt-6 px-4">
   610→            <div className={`rounded-lg border p-4 text-sm ${isDark ? "bg-yellow-900/20 border-yellow-800 text-yellow-300" : "bg-yellow-50 border-yellow-200 text-yellow-700"}`}>
   611→              {data.warning}
   612→            </div>
   613→          </div>
   614→        )}
   615→        <div className="flex flex-col items-center">
   616→          {displayImages.map((src, i) => (
   617→            <div key={i} className="relative w-full" style={i === 0 ? { marginTop: data.warning ? "0.5rem" : "0" } : undefined}>
   618→              {!loadedImages.has(i) && (
   619→                <Skeleton className="w-full mx-auto" style={{ aspectRatio: "2/3", maxWidth: "800px" }} />
   620→              )}
   621→              <img ref={(el) => { if (el) imgRefs.current.set(i, el); }}
   622→                src={src} alt={`Page ${i + 1}`} onLoad={() => handleImageLoad(i)}
   623→                style={{ ...imgStyle, display: loadedImages.has(i) ? "block" : "none" }}
   624→                loading="eager" />
   625→            </div>
   626→          ))}
   627→        </div>
   628→        <div className="h-20" />
   629→      </div>
   630→
   631→      {/* Bottom nav */}
   632→      <nav className={`shrink-0 border-t px-4 py-2 flex items-center justify-between ${barBg} z-40`}>
   633→        <Button variant="ghost" size="sm" disabled={!data.prevUrl} onClick={() => goToChapter(data.prevUrl)}>
   634→          <ChevronLeft className="h-4 w-4 mr-1" /><span className="hidden sm:inline">Prev</span>
   635→        </Button>
   636→
   637→        <div className="flex items-center gap-2">
   638→          <button onClick={() => setShowPageJumper(true)} className={`px-2 py-1 rounded text-xs ${subText} ${hoverBg}`}>
   639→            {currentPage}/{totalPages}
   640→          </button>
   641→        </div>
   642→
   643→        <Button variant="ghost" size="sm" disabled={!data.nextUrl} onClick={() => goToChapter(data.nextUrl)}>
   644→          <span className="hidden sm:inline">Next</span><ChevronRight className="h-4 w-4 ml-1" />
   645→        </Button>
   646→      </nav>
   647→
   648→      {/* Scroll to top */}
   649→      {showScrollTop && (
   650→        <button onClick={scrollToTop}
   651→          className={`fixed bottom-20 right-4 p-2.5 rounded-full shadow-lg transition-all z-30 ${isDark ? "bg-white/10 hover:bg-white/20 text-white" : "bg-white hover:bg-gray-100 text-gray-900 border"}`}
   652→          aria-label="Scroll to top"><ArrowUp className="h-5 w-5" /></button>
   653→      )}
   654→    </div>
   655→  );
   656→}
