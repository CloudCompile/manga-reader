    64→    64→function getHistory(): HistoryEntry[] {

    67→    67→function saveHistory(entry: HistoryEntry) {
    68→    68→  const h = getHistory().filter((e) => e.url !== entry.url);
    69→    69→  h.unshift({ ...entry, timestamp: Date.now() });
    70→    70→  localStorage.setItem("manga-history", JSON.stringify(h.slice(0, 50)));
    71→    71→}

    79→    79→// ─── IndexedDB for offline ───────────────────────────
    80→    80→const DB_NAME = "manga-offline";
    81→    81→const DB_VERSION = 1;
    82→    82→const STORE = "chapters";
    83→    83→
    84→    84→function openDB(): Promise<IDBDatabase> {
    85→    85→  return new Promise((resolve, reject) => {
    86→    86→    const req = indexedDB.open(DB_NAME, DB_VERSION);
    87→    87→    req.onupgradeneeded = () => req.result.createObjectStore(STORE, { keyPath: "url" });
    88→    88→    req.onsuccess = () => resolve(req.result);
    89→    89→    req.onerror = () => reject(req.error);
    90→    90→  });
    91→    91→}
    92→    92→
    93→    93→async function saveOffline(data: OfflineChapter) {
    94→    94→  const db = await openDB();
    95→    95→  return new Promise<void>((resolve, reject) => {
    96→    96→    const tx = db.transaction(STORE, "readwrite");
    97→    97→    tx.objectStore(STORE).put(data);
    98→    98→    tx.oncomplete = () => resolve();
    99→    99→    tx.onerror = () => reject(tx.error);
   100→   100→  });
   101→   101→}
   102→   102→
   103→   103→async function getOffline(url: string): Promise<OfflineChapter | null> {
   104→   104→  const db = await openDB();
   105→   105→  return new Promise((resolve, reject) => {
   106→   106→    const tx = db.transaction(STORE, "readonly");
   107→   107→    const req = tx.objectStore(STORE).get(url);
   108→   108→    req.onsuccess = () => resolve(req.result || null);
   109→   109→    req.onerror = () => reject(req.error);
   110→   110→  });
   111→   111→}
   112→   112→
   113→   113→async function getAllOffline(): Promise<OfflineChapter[]> {
   114→   114→  const db = await openDB();
   115→   115→  return new Promise((resolve, reject) => {
   116→   116→    const tx = db.transaction(STORE, "readonly");
   117→   117→    const req = tx.objectStore(STORE).getAll();
   118→   118→    req.onsuccess = () => resolve(req.result || []);
   119→   119→    req.onerror = () => reject(req.error);
   120→   120→  });
   121→   121→}

   123→   123→async function deleteOffline(url: string) {
   124→   124→  const db = await openDB();
   125→   125→  return new Promise<void>((resolve, reject) => {
   126→   126→    const tx = db.transaction(STORE, "readwrite");
   127→   127→    tx.objectStore(STORE).delete(url);
   128→   128→    tx.oncomplete = () => resolve();
   129→   129→    tx.onerror = () => reject(tx.error);
   130→   130→  });
   131→   131→}
